One-time Supabase setup

Run this SQL in your Supabase SQL Editor once to create the table, then collapse this.

create table if not exists whispers_posts (
  id          text primary key default gen_random_uuid()::text,
  is_page     boolean default true,
  page_name   text not null,
  page_avatar text not null,
  page_initial text not null,
  text        text not null,
  image_url   text,
  is_poll     boolean default false,
  poll_options jsonb,
  likes       integer default 0,
  comments    integer default 0,
  created_at  timestamptz default now()
);

-- allow anyone to read, only service role to insert/delete
-- (anon key is used here so disable RLS or add a permissive policy):
alter table whispers_posts enable row level security;
create policy "public read"  on whispers_posts for select using (true);
create policy "admin insert" on whispers_posts for insert with check (true);
create policy "admin delete" on whispers_posts for delete using (true);
